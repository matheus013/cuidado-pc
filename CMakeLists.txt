cmake_minimum_required(VERSION 3.21)
project(CuidadoPC VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent)
qt_standard_project_setup()
qt_add_executable(cuidado-pc src/main.cpp src/backend.cpp src/backend.h src/cleanup.cpp src/cleanup.h)
qt_add_resources(cuidado-pc "app_resources" PREFIX "/" FILES qml/Main.qml)
target_link_libraries(cuidado-pc PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
set_target_properties(cuidado-pc PROPERTIES MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
    qt_add_executable(cleanup-tests tests/test_cleanup.cpp src/cleanup.cpp src/cleanup.h)
    target_include_directories(cleanup-tests PRIVATE src)
    target_link_libraries(cleanup-tests PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME cleanup-tests COMMAND cleanup-tests)
endif()
